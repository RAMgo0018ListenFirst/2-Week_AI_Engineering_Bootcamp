Q1.In Python, a list is ordered and changeable, so we can add or remove items from it. I would use a list when storing things like marks of students because the values may change later.
A tuple is also ordered but cannot be changed after creation, so it is useful for fixed data like coordinates (x, y).
A set stores only unique values and does not keep order, so I would use it when removing duplicates from a list.
A dict stores data in key:value form, which helps in connecting related data together. I would use a dict for storing student details like {"name": "Ram", "age": 20} because values can be accessed using keys.

    try:

            data = json.load(file)

    except FileNotFoundError:
        return {}

import requests
from dotenv import load_dotenv
# Load variables from .env file
load_dotenv()

# GitHub API url
url = f"https://api.github.com/users/anthropics?api_key={GITHUB_API_KEY}"
try:
    response = requests.get(url)
    # Check if request was successful
    if response.status_code == 200:
        data = response.json()
        # Print required fields
        print("Name:", data.get("name"))
        print("Public Repos:", data.get("public_repos"))
        print("Followers:", data.get("followers"))
    else:
        # Handle failed requests
        print("Error: Request failed")
except requests.exceptions.RequestException as e:
    # Handle request-related errors
    print("Error while making request:", e)

Q9.With normal synchronous code, the program waits for one task to finish completely before doing anything else.
With async code, the program can work on other tasks while waiting for slow things like APIs or databases.
import httpx
import asyncio
async def fetch(url):
    # Here we used httpx and .client to get request like we do in requests but difference is here it work better with async await things
    async with httpx.AsyncClient() as client:
        response = await client.get(url, timeout=10)
        response.raise_for_status()#This basically raise error incase of incorrect statuscodes
        return response.json()
async def main():
    data = await fetch("https://api.github.com/users/anthropics")
    print(data)
asyncio.run(main())

Q14.import { useState } from "react";
function Counter() {
  const [count, setCount] = useState(0);
  // Increase count by 1
  function increment() {
    setCount(count + 1);
  }
  function decrement() {
    if (count > 0) {
      setCount(count - 1);
    }
  }
  function reset() {
    setCount(0);
  }
  return (
    <div>
      <h2>Count: {count}</h2>
      <button onClick={increment}>+1</button>
      <button onClick={decrement}>-1</button>
      <button onClick={reset}>Reset</button>
    </div>
  );
}

export default Counter;

Q16 . import { useEffect, useState } from "react";

function SearchableList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [search, setSearch] = useState("");
  useEffect(() => {

    fetch("https://jsonplaceholder.typicode.com/users")
      .then((response) => {

        // Check if request failed
        if (!response.ok) {
          throw new Error("Failed to fetch users");
        }

        return response.json();
      })
      .then((data) => {
        setUsers(data);
        setLoading(false);
      })
      .catch((err) => {
        setError(err.message);
        setLoading(false);
      });

  }, []);
  const filteredUsers = users.filter((user) =>
    user.name.toLowerCase().includes(search.toLowerCase())
  );

  // Loading UI
  if (loading) {
    return <h2>Loading...</h2>;
  }

  // Error UI
  if (error) {
    return <h2>Error: {error}</h2>;
  }

  return (
    <div>
      <input
        type="text"
        placeholder="Search users"
        value={search}
        onChange={(e) => setSearch(e.target.value)}
      />
      {filteredUsers.length > 0 ? (
        <ul>
          {filteredUsers.map((user) => (
            <li key={user.id}>
              {user.name}
            </li>
          ))}
        </ul>
      ) : (
        <p>No users found</p>
      )}

    </div>
  );
}
export default SearchableList;
